Overview

For this Hackathon project, we decided to look into electric vehicle adoption and public charging infrastructure across all the 16 counties in the DFW region (as designated by the NCTCOG). Data sources are provided by DFW Clean Cities, Alternative Fuels Data Center (AFDC), and the Census Bureau. Posit Connect Cloud was used to properly implement the Shiny application because running it via Quarto did not work as intended.

## Counting charging stations by county
dfw_ev_stations_counts <- dfw_ev_stations %>%
  st_drop_geometry() %>%
  group_by(County) %>%
  summarise(
    charging_stations = n_distinct(ID),
    .groups = "drop"
  ) %>%
  arrange(desc(charging_stations))

### nrow(dfw_ev_stations)
### sum(dfw_ev_stations_counts$charging_stations)

# Counting charging ports by county
dfw_ev_ports_counts <- dfw_ev_stations %>%
  st_drop_geometry() %>%
  mutate(
    level1_ports = as.numeric(`EV Level1 EVSE Num`),
    level2_ports = as.numeric(`EV Level2 EVSE Num`),
    dc_fast_ports = as.numeric(`EV DC Fast Count`)
  ) %>%
  group_by(County) %>%
  summarise(
    charging_ports = sum(
      level1_ports,
      level2_ports,
      dc_fast_ports,
      na.rm = TRUE
    ),
    .groups = "drop"
  ) %>%
  arrange(desc(charging_ports))

ev_font <- "quattrocento"
ev_colors <- c(
  primary = "purple4",
  secondary = "darkorchid3",
  accent = "violet",
  text = "grey15",
  muted = "grey35",
  grid = "grey90"
)


ev_theme <- function(){
  theme_minimal(
    base_family = ev_font,
    base_size = 16
  ) +
    theme(
      # for chart title
      plot.title = element_text(
        face = "bold",
        size = 25,
        hjust = 0.5,
        color = ev_colors[["text"]]
      ),
      # for chart subtitle
      plot.subtitle = element_text(
        size = 15,
        hjust = 0.5,
        color = ev_colors[["muted"]]
      ),
      # for sources
      plot.caption = element_text(
        size = 12,
        hjust = 0,
        color = ev_colors[["muted"]]
      ),
      # for X-Y axes
      axis.title = element_text(
        color = ev_colors[["text"]]
      ),
      axis.text = element_text(
        size = 15,
        color = ev_colors[["muted"]]
      ),
      # for grid lines
      panel.grid.major = element_line(
        color = ev_colors[["grid"]]
      ),
      panel.grid.minor = element_blank(),
      # for legend position
      legend.position = "bottom"
    )
}

county_metrics <- dfw_county_adoption %>%
  left_join(
    dfw_ev_stations_counts,
    by = "County"
  ) %>%
  left_join(
    dfw_ev_ports_counts,
    by = "County"
  ) %>%
  transmute(
    County,
    ev_registrations = `Electric Vehicles`,
    charging_stations = coalesce(charging_stations, 0L),
    charging_ports = coalesce(charging_ports, 0),
    ev_adoption_rate = `Adoption Rate` * 100
  ) %>%
  arrange(County)

# ============================================================
# 4. RESHAPE DATA
# ============================================================

chart_actual <- county_metrics %>%
  pivot_longer(
    cols = c(
      ev_registrations,
      charging_stations,
      charging_ports,
      ev_adoption_rate
    ),
    names_to = "metric",
    values_to = "actual_value"
  ) %>%
  mutate(
    metric = recode(
      metric,
      ev_registrations = "EV Registrations",
      charging_stations = "Charging Stations",
      charging_ports = "Charging Ports",
      ev_adoption_rate = "EV Adoption Rate (%)"
    )
  )

# ============================================================
# 5. NORMALIZE VALUES TO 0–100
# ============================================================

chart_data <- chart_actual %>%
  group_by(metric) %>%
  mutate(
    normalized_value = if (
      any(is.finite(actual_value)) &&
      max(actual_value, na.rm = TRUE) > 0
    ) {
      actual_value /
        max(actual_value, na.rm = TRUE) * 100
    } else {
      0
    }
  ) %>%
  ungroup()


# ============================================================
# 6. PREPARE CIRCULAR BAR POSITIONS
# ============================================================

county_order <- dfw_counties

metric_order <- c(
  "EV Registrations",
  "Charging Stations",
  "Charging Ports",
  "EV Adoption Rate (%)"
)

bars_per_county <- length(metric_order)
gap_size <- 2

chart_data <- chart_data %>%
  mutate(
    County = factor(County, levels = county_order),
    metric = factor(metric, levels = metric_order)
  ) %>%
  arrange(County, metric) %>%
  group_by(County) %>%
  mutate(
    metric_id = row_number()
  ) %>%
  ungroup() %>%
  mutate(
    county_id = as.integer(County),
    
    bar_id = (county_id - 1) *
      (bars_per_county + gap_size) + metric_id,
    
    bar_height = normalized_value
  )

total_positions <- length(county_order) *
  (bars_per_county + gap_size)

inner_radius <- 30


# ============================================================
# 7. DEFINE PURPLE COLOR PALETTE
# ============================================================

metric_colors <- c(
  "EV Registrations" = "purple4",
  "Charging Stations" = "darkorchid3",
  "Charging Ports" = "violet",
  "EV Adoption Rate (%)" = "thistle2"
)


# ============================================================
# 8. CALCULATE COUNTY LABEL POSITIONS
# ============================================================

county_labels <- chart_data %>%
  group_by(County) %>%
  summarise(
    label_position = mean(bar_id),
    .groups = "drop"
  ) %>%
  mutate(
    angle = 90 -
      360 * (label_position - 0.5) / total_positions,
    
    hjust = ifelse(angle < -90, 1, 0),
    
    angle = ifelse(
      angle < -90,
      angle + 180,
      angle
    )
  )

sector_width <- bars_per_county + gap_size

county_dividers <- data.frame(
  boundary = (seq_along(county_order) - 1) *
    sector_width +
    bars_per_county +
    gap_size / 2 + 0.5
)

# ============================================================
# 9A. CREATE CIRCULAR CHART - RIGHT SIDE
# ============================================================

circular_plot <- ggplot(
  chart_data,
  aes(
    x = bar_id,
    y = bar_height,
    fill = metric
  )
) +
  
  geom_segment(
    data = county_dividers,
    aes(
      x = boundary,
      xend = boundary,
      y = 0,
      yend = 110
    ),
    inherit.aes = FALSE,
    color = "grey92",
    linewidth = 0.3
  ) +
  
  geom_col(
    width = 0.92,
    color = "white",
    linewidth = 0.2
  ) +
  
  geom_text(
    data = county_labels,
    aes(
      x = label_position,
      y = 116,
      label = County,
      angle = angle,
      hjust = hjust
    ),
    inherit.aes = FALSE,
    size = 4,
    family = ev_font,
    color = "grey15",
    fontface = "bold"
  ) +
  
  coord_polar(
    theta = "x",
    start = 0,
    clip = "off"
  ) +
  
  scale_y_continuous(
    limits = c(-inner_radius, 130),
    breaks = c(0, 25, 50, 75, 100),
    expand = c(0, 0)
  ) +
  
  scale_x_continuous(
    limits = c(0.5, total_positions + 0.5),
    breaks = NULL,
    expand = c(0, 0)
  ) +
  
  scale_fill_manual(
    values = metric_colors,
    name = NULL
  ) +
  
  ev_theme() +
  
  theme(
    axis.text = element_blank(),

  filtered_data <- reactive({
    dashboard_data
  })
  
  # Chart 9
  output$chart9_output <- renderPlotly({
    chart9_filtered <- chart9 %+% filtered_data() +
      labs(
        title = NULL,

      chart9_filtered,

      plotly::layout(

        xaxis = list(

          tickfont = list(size = 12)
        ),
        yaxis = list(
          title = list(font = list(size = 15)),
          tickfont = list(size = 12)
      )
    )
  })
  
  # Chart 18
  output$chart18_output <- renderPlotly({
    chart18_filtered <- chart18 %+% filtered_data() +

  output$chart5_output <- renderPlot({

  }, res = 96)
